-- 2. CREATE TABLES
-- ============================================

-- TABLE 1: Users
CREATE TABLE users (
    user_id BIGSERIAL PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NULL,
    supabase_user_id UUID NULL UNIQUE,
    role VARCHAR(20) NOT NULL DEFAULT 'Citizen' CHECK (role IN ('Citizen', 'Municipal')),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    last_login TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_supabase_id ON users(supabase_user_id);
CREATE INDEX idx_users_role ON users(role);

-- TABLE 2: CitizenProfiles
CREATE TABLE citizen_profiles (
    citizen_profile_id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone_number VARCHAR(20) NULL,
    address_line VARCHAR(255) NULL,
    city VARCHAR(100) NULL,
    latitude DECIMAL(10,7) NULL,
    longitude DECIMAL(10,7) NULL,
    notification_preference VARCHAR(20) DEFAULT 'Push',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

CREATE INDEX idx_citizen_profiles_user_id ON citizen_profiles(user_id);

-- TABLE 3: MunicipalUsers
CREATE TABLE municipal_users (
    municipal_user_id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    employee_id VARCHAR(50) NOT NULL UNIQUE,
    department VARCHAR(100) NOT NULL,
    position_title VARCHAR(100) NOT NULL,
    hire_date DATE NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW(),
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE
);

CREATE INDEX idx_municipal_users_user_id ON municipal_users(user_id);
CREATE INDEX idx_municipal_users_employee ON municipal_users(employee_id);

-- TABLE 4: FaultCategories
CREATE TABLE fault_categories (
    category_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description VARCHAR(255) NULL,
    severity_weight DECIMAL(4,2) NOT NULL CHECK (severity_weight BETWEEN 1 AND 10),
    estimated_fix_time INT NULL,
    icon VARCHAR(50) NULL,
    is_active BOOLEAN DEFAULT TRUE
);

-- TABLE 5: EssentialServices
CREATE TABLE essential_services (
    service_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    service_type VARCHAR(50) NOT NULL CHECK (service_type IN ('School', 'Clinic', 'Hospital', 'Police', 'Fire Station')),
    address_line VARCHAR(255) NULL,
    latitude DECIMAL(10,7) NOT NULL,
    longitude DECIMAL(10,7) NOT NULL,
    ward_number VARCHAR(10) NULL,
    contact_number VARCHAR(20) NULL
);

CREATE INDEX idx_essential_services_lat_long ON essential_services(latitude, longitude);

-- TABLE 6: ServiceAreas
CREATE TABLE service_areas (
    area_id BIGSERIAL PRIMARY KEY,
    area_name VARCHAR(50) NOT NULL UNIQUE,
    ward_number VARCHAR(10) NOT NULL,
    description VARCHAR(255) NULL,
    latitude DECIMAL(10,7) NULL,
    longitude DECIMAL(10,7) NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_service_areas_ward ON service_areas(ward_number);

-- TABLE 7: FaultReports
CREATE TABLE fault_reports (
    report_id BIGSERIAL PRIMARY KEY,
    citizen_id BIGINT NOT NULL,
    category_id BIGINT NOT NULL,
    area_id BIGINT NULL,
    title VARCHAR(200) NOT NULL,
    description VARCHAR(1000) NOT NULL,
    latitude DECIMAL(10,7) NOT NULL,
    longitude DECIMAL(10,7) NOT NULL,
    address_line VARCHAR(255) NULL,
    status VARCHAR(30) DEFAULT 'Pending' CHECK (status IN ('Pending', 'Verified', 'InProgress', 'Resolved', 'Rejected')),
    priority_score INT NULL CHECK (priority_score BETWEEN 0 AND 100),
    reference_number VARCHAR(20) NOT NULL UNIQUE,
    verified_by BIGINT NULL,
    verified_at TIMESTAMP NULL,
    assigned_to BIGINT NULL,
    resolution_notes VARCHAR(500) NULL,
    resolved_at TIMESTAMP NULL,
    is_duplicate BOOLEAN DEFAULT FALSE,
    is_offline BOOLEAN DEFAULT FALSE,
    sync_status VARCHAR(20) DEFAULT 'Synced' CHECK (sync_status IN ('Pending', 'Synced', 'Failed')),
    offline_id VARCHAR(50) NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (citizen_id) REFERENCES citizen_profiles(citizen_profile_id),
    FOREIGN KEY (category_id) REFERENCES fault_categories(category_id),
    FOREIGN KEY (area_id) REFERENCES service_areas(area_id),
    FOREIGN KEY (verified_by) REFERENCES municipal_users(municipal_user_id),
    FOREIGN KEY (assigned_to) REFERENCES municipal_users(municipal_user_id)
);

CREATE INDEX idx_fault_reports_citizen ON fault_reports(citizen_id);
CREATE INDEX idx_fault_reports_category ON fault_reports(category_id);
CREATE INDEX idx_fault_reports_status ON fault_reports(status);
CREATE INDEX idx_fault_reports_created ON fault_reports(created_at);
CREATE INDEX idx_fault_reports_reference ON fault_reports(reference_number);

-- TABLE 8: ReportPhotos
CREATE TABLE report_photos (
    photo_id BIGSERIAL PRIMARY KEY,
    report_id BIGINT NOT NULL,
    photo_url VARCHAR(500) NOT NULL,
    supabase_storage_path VARCHAR(500) NULL,
    is_primary BOOLEAN DEFAULT FALSE,
    is_offline BOOLEAN DEFAULT FALSE,
    sync_status VARCHAR(20) DEFAULT 'Synced' CHECK (sync_status IN ('Pending', 'Synced', 'Failed')),
    offline_photo_path VARCHAR(500) NULL,
    uploaded_at TIMESTAMP DEFAULT NOW(),
    FOREIGN KEY (report_id) REFERENCES fault_reports(report_id) ON DELETE CASCADE
);

CREATE INDEX idx_report_photos_report ON report_photos(report_id);

-- TABLE 9: DuplicateReports
CREATE TABLE duplicate_reports (
    duplicate_link_id BIGSERIAL PRIMARY KEY,
    original_report_id BIGINT NOT NULL,
    duplicate_report_id BIGINT NOT NULL,
    match_percentage DECIMAL(5,2) NOT NULL,
    marked_by BIGINT NULL,
    marked_at TIMESTAMP DEFAULT NOW(),
    is_merged BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (original_report_id) REFERENCES fault_reports(report_id),
    FOREIGN KEY (duplicate_report_id) REFERENCES fault_reports(report_id),
    FOREIGN KEY (marked_by) REFERENCES municipal_users(municipal_user_id),
    CONSTRAINT different_reports CHECK (original_report_id != duplicate_report_id)
);

CREATE INDEX idx_duplicate_original ON duplicate_reports(original_report_id);
CREATE INDEX idx_duplicate_duplicate ON duplicate_reports(duplicate_report_id);

-- TABLE 10: PriorityScores
CREATE TABLE priority_scores (
    score_id BIGSERIAL PRIMARY KEY,
    report_id BIGINT NOT NULL UNIQUE,
    total_score DECIMAL(6,2) NOT NULL CHECK (total_score BETWEEN 0 AND 100),
    severity_factor DECIMAL(6,2) NOT NULL CHECK (severity_factor BETWEEN 0 AND 25),
    proximity_factor DECIMAL(6,2) NOT NULL CHECK (proximity_factor BETWEEN 0 AND 35),
    duplicate_factor DECIMAL(6,2) NOT NULL CHECK (duplicate_factor BETWEEN 0 AND 20),
    age_factor DECIMAL(6,2) NOT NULL CHECK (age_factor BETWEEN 0 AND 20),
    computed_at TIMESTAMP DEFAULT NOW(),
    FOREIGN KEY (report_id) REFERENCES fault_reports(report_id) ON DELETE CASCADE
);

CREATE INDEX idx_priority_scores_report ON priority_scores(report_id);

-- TABLE 11: FaultStatusHistory
CREATE TABLE fault_status_history (
    history_id BIGSERIAL PRIMARY KEY,
    report_id BIGINT NOT NULL,
    old_status VARCHAR(30) NULL,
    new_status VARCHAR(30) NOT NULL,
    changed_by BIGINT NOT NULL,
    changed_at TIMESTAMP DEFAULT NOW(),
    notes VARCHAR(255) NULL,
    FOREIGN KEY (report_id) REFERENCES fault_reports(report_id) ON DELETE CASCADE,
    FOREIGN KEY (changed_by) REFERENCES municipal_users(municipal_user_id)
);

CREATE INDEX idx_history_report ON fault_status_history(report_id);

-- TABLE 12: ReportFeedback
CREATE TABLE report_feedback (
    feedback_id BIGSERIAL PRIMARY KEY,
    report_id BIGINT NOT NULL UNIQUE,
    citizen_id BIGINT NOT NULL,
    rating INT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment VARCHAR(500) NULL,
    response_time INT NULL,
    is_anonymous BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (report_id) REFERENCES fault_reports(report_id) ON DELETE CASCADE,
    FOREIGN KEY (citizen_id) REFERENCES citizen_profiles(citizen_profile_id)
);

CREATE INDEX idx_feedback_report ON report_feedback(report_id);

-- TABLE 13: Notifications
CREATE TABLE notifications (
    notification_id BIGSERIAL PRIMARY KEY,
    citizen_id BIGINT NOT NULL,
    report_id BIGINT NULL,
    notification_type VARCHAR(50) CHECK (notification_type IN ('StatusUpdate', 'Assignment', 'Reminder', 'Alert')),
    channel VARCHAR(20) DEFAULT 'Push' CHECK (channel IN ('Email', 'SMS', 'Push')),
    subject VARCHAR(200) NOT NULL,
    message VARCHAR(500) NOT NULL,
    device_token VARCHAR(255) NULL,
    fcm_message_id VARCHAR(100) NULL,
    is_sent BOOLEAN DEFAULT FALSE,
    sent_at TIMESTAMP NULL,
    is_read BOOLEAN DEFAULT FALSE,
    read_at TIMESTAMP NULL,
    is_delivered BOOLEAN DEFAULT FALSE,
    delivered_at TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    FOREIGN KEY (citizen_id) REFERENCES citizen_profiles(citizen_profile_id),
    FOREIGN KEY (report_id) REFERENCES fault_reports(report_id) ON DELETE SET NULL

CREATE INDEX idx_notifications_citizen ON notifications(citizen_id);

-- ============================================
-- 4. RLS POLICIES
-- ============================================

CREATE POLICY "Users can read own profile" ON users
    FOR SELECT USING (supabase_user_id = auth.uid());

CREATE POLICY "Citizens can manage own profile" ON citizen_profiles
    FOR ALL USING (user_id IN (SELECT user_id FROM users WHERE supabase_user_id = auth.uid()));

CREATE POLICY "Municipal can read all reports" ON fault_reports
    FOR SELECT USING (EXISTS (SELECT 1 FROM users WHERE supabase_user_id = auth.uid() AND role = 'Municipal'));

CREATE POLICY "Citizens can insert reports" ON fault_reports
    FOR INSERT WITH CHECK (EXISTS (SELECT 1 FROM users WHERE supabase_user_id = auth.uid() AND role = 'Citizen'));

CREATE POLICY "Citizens can read own reports" ON fault_reports
    FOR SELECT USING (citizen_id IN (SELECT citizen_profile_id FROM citizen_profiles WHERE user_id IN (SELECT user_id FROM users WHERE supabase_user_id = auth.uid())));

-- ============================================
-- 5. FUNCTION: Generate Report Reference Number
-- ============================================
CREATE OR REPLACE FUNCTION generate_report_id()
RETURNS VARCHAR(20) AS $$
DECLARE
    random_num INT;
    result VARCHAR(20);
BEGIN
    random_num := floor(random() * 10000)::INT;
    result := 'FIX-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || LPAD(random_num::TEXT, 4, '0');
    RETURN result;
END;
$$ LANGUAGE plpgsql;

-- ============================================
-- 6. STORED PROCEDURE: Calculate Priority Score
-- ============================================
CREATE OR REPLACE PROCEDURE calculate_priority_score(p_report_id BIGINT)
LANGUAGE plpgsql
AS $$
DECLARE
    v_severity DECIMAL(6,2);
    v_proximity DECIMAL(6,2);
    v_duplicate DECIMAL(6,2);
    v_age DECIMAL(6,2);
    v_total DECIMAL(6,2);
    v_hours_old INT;
BEGIN
    SELECT fc.severity_weight * 2.5 INTO v_severity
    FROM fault_reports fr
    JOIN fault_categories fc ON fr.category_id = fc.category_id
    WHERE fr.report_id = p_report_id;

    v_proximity := 20.0;

    SELECT CASE 
        WHEN COUNT(*) > 0 THEN 20.0
        ELSE 0.0
    END INTO v_duplicate
    FROM duplicate_reports
    WHERE original_report_id = p_report_id OR duplicate_report_id = p_report_id;

    SELECT EXTRACT(HOUR FROM (NOW() - created_at)) INTO v_hours_old
    FROM fault_reports
    WHERE report_id = p_report_id;
    
    v_age := CASE 
        WHEN v_hours_old >= 48 THEN 20.0
        WHEN v_hours_old >= 24 THEN 15.0
        WHEN v_hours_old >= 12 THEN 10.0
        WHEN v_hours_old >= 6 THEN 5.0
        ELSE 0.0
    END;

    v_total := v_severity + v_proximity + v_duplicate + v_age;

    INSERT INTO priority_scores (report_id, total_score, severity_factor, proximity_factor, duplicate_factor, age_factor)
    VALUES (p_report_id, v_total, v_severity, v_proximity, v_duplicate, v_age)
    ON CONFLICT (report_id) DO UPDATE SET
        total_score = EXCLUDED.total_score,
        severity_factor = EXCLUDED.severity_factor,
        proximity_factor = EXCLUDED.proximity_factor,
        duplicate_factor = EXCLUDED.duplicate_factor,
        age_factor = EXCLUDED.age_factor,
        computed_at = NOW();

    UPDATE fault_reports SET priority_score = v_total WHERE report_id = p_report_id;
END;
$$;
